-- CORRECCIÓN DE RESERVAS DE POOL -- Mantiene los datos existentes y deja exactamente las mesas 1, 2 y 3 para reservas de pool. UPDATE public.reservas SET mesa_pool = NULLIF( (regexp_match(COALESCE(notas, ''), '\[MESA_POOL:([1-3])\]'))[1], '' )::integer WHERE servicio = 'Pool' AND mesa_pool IS NULL AND notas ~ '\[MESA_POOL:[1-3]\]'; CREATE UNIQUE INDEX IF NOT EXISTS reservas_pool_slot_unico ON public.reservas (fecha, hora, mesa_pool) WHERE servicio = 'Pool' AND mesa_pool IS NOT NULL AND estado <> 'cancelada'; DROP POLICY IF EXISTS "public can create reservations" ON public.reservas; CREATE POLICY "public can create reservations" ON public.reservas FOR INSERT TO anon, authenticated WITH CHECK ( length(trim(COALESCE(nombre, ''))) BETWEEN 2 AND 100 AND length(trim(COALESCE(telefono, ''))) BETWEEN 6 AND 40 AND fecha IS NOT NULL AND hora IS NOT NULL AND length(trim(COALESCE(servicio, ''))) BETWEEN 2 AND 100 AND COALESCE(cantidad_personas, 1) BETWEEN 1 AND 50 AND ( servicio <> 'Pool' OR mesa_pool IN (1, 2, 3) ) );
